#include <iostream>
#include <string>

using namespace std;

int ReadNumber(string Message)
{
    int Number;

    cout << Message << endl;
    cin >> Number;

    return Number;
}

float SumNumbers()
{
    int Sum = 0;
    int Number = 0;
    int Counter = 1;

    while (true)
    {
        Number = ReadNumber("Please Enter Number (" + to_string(Counter) + ") or -99 to stop");

        if (Number == -99)
            break;

        Sum += Number;
        Counter++;
    }

    return Sum;
}

int main()
{
    cout << "Result is: " << SumNumbers();

    return 0;
}
