#include <iostream>

using namespace std;

float ReadDiameter()
{
    float D;
    cout << "Please Enter Diameter D?\n";
    cin >> D;

    return D;
}

float CircleAreaFromDiameter(float D)
{
    const float PI = 3.14f;

    return (PI * D * D) / 4;
}

void PrintResult(float Area)
{
    cout << "\nCircle Area: " << Area << endl;
}

int main()
{
    PrintResult(CircleAreaFromDiameter(ReadDiameter()));

    return 0;
}
