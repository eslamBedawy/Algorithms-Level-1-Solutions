#include <iostream>
#include <cmath>

using namespace std;

void ReadNumber(float& A, float& D)
{
    cout << "Please Enter Rectangle Side A?\n";
    cin >> A;

    cout << "Please Enter Rectangle Diagonal D?\n";
    cin >> D;
}

float CalculateRectangleAreaFromSideAndDiagonal(float A, float D)
{
    float Area = A * sqrt((D * D) - (A * A));
    return Area;
}

void PrintResult(float Area)
{
    cout << "\nRectangle Area: " << Area << endl;
}

int main()
{
    float A, D;

    ReadNumber(A, D);

    PrintResult(CalculateRectangleAreaFromSideAndDiagonal(A, D));

    return 0;
}
