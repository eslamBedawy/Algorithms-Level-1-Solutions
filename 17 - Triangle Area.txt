#include <iostream>

using namespace std;

void ReadNumber(float& Base, float& Height)
{
    cout << "Please Enter Triangle Base?\n";
    cin >> Base;

    cout << "Please Enter Triangle Height?\n";
    cin >> Height;
}

float TriangleArea(float Base, float Height)
{
    return (Base * Height) / 2;
}

void PrintResult(float Area)
{
    cout << "\nTriangle Area: " << Area << endl;
}

int main()
{
    float Base, Height;

    ReadNumber(Base, Height);

    PrintResult(TriangleArea(Base, Height));

    return 0;
}
