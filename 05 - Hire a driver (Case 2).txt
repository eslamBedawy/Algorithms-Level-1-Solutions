#include <iostream>
#include <string>

using namespace std;

struct stInfo
{
    int Age;
    bool HasDrivingLicense;
    bool HasRecommendation;
};

stInfo ReadInfo()
{
    stInfo Info;

    cout << "Please Enter Your Age?\n";
    cin >> Info.Age;

    cout << "Do You Have a Driving License? (1 = Yes, 0 = No)\n";
    cin >> Info.HasDrivingLicense;

    cout << "Do You Have a Recommendation? (1 = Yes, 0 = No)\n";
    cin >> Info.HasRecommendation;

    return Info;
}

bool IsAccepted(stInfo Info)
{
    if (Info.HasRecommendation)
        return true;
    else
        return (Info.Age > 21 && Info.HasDrivingLicense);
}

void PrintResult(stInfo Info)
{
    if (IsAccepted(Info))
        cout << "Hired\n";
    else
        cout << "Rejected\n";
}

int main()
{
    PrintResult(ReadInfo());
    return 0;
}
