#include <iostream>
#include <string>

using namespace std;

string ReadPinCode()
{
    string PinCode;

    cout << "Please Enter Pin Code?\n";
    cin >> PinCode;

    return PinCode;
}

bool Login()
{
    string PinCode;

    for (int i = 0; i < 3; i++)
    {
        PinCode = ReadPinCode();

        if (PinCode == "1234")
        {
            return true;
        }
        else
        {
            cout << "Wrong PIN!!\n";
            system("color 4");
        }
    }

    return false;
}

int main()
{
    if (Login())
    {
        system("color 2");
        cout << "Your Account Balance is: " << 7500 << endl;
    }
    else
    {
        cout << "Access Denied!\n";
    }

    return 0;
}
