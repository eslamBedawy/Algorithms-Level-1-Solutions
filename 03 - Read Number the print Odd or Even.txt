#include <iostream>
#include <string>

using namespace std;

enum enCheckOddOrEven { Even = 1, Odd = 2 };

int ReadNumber()
{
    int Number;
    cout << "Please Enter Number? ";
    cin >> Number;

    return Number;
}

enCheckOddOrEven CheckOddOrEven(int Number)
{
    int Result = Number % 2;

    if (Result == 0)
        return enCheckOddOrEven::Even;
    else
        return enCheckOddOrEven::Odd;
}

void PrintResult(int Number)
{
    if (CheckOddOrEven(Number) == enCheckOddOrEven::Even)
        cout << "Number is Even\n";
    else
        cout << "Number is Odd\n";
}

int main()
{
    PrintResult(ReadNumber());
    return 0;
}
